# 一次性 py 脚本：把 line=男频 / line=混合 全部刷为 line=古言仙侠
import os, re
chron = 'seasons/01-xianxia/chronicle'
for fn in os.listdir(chron):
    p = os.path.join(chron, fn)
    if not fn.startswith('ch') or not fn.endswith('.md'): continue
